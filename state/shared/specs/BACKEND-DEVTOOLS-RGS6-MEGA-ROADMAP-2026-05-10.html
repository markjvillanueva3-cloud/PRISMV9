#7.3 Synergy classifier rubric

features = {
  semantic_match_score: embed(content) × embed(PRISM substrate)
  duplication_risk:     duplicationGuardEngine.checkBeforeCreating()
  domain_match:         mapToL5domain(content)
  ai_priority_score:    ai-priority-rank.mjs
  effort_estimate:      tokens(content) → heuristic hours
  blast_radius:         peer_claim_count + file_count
  novelty_strength:     1 - max(cosine to known items)
}

decision = high if semantic_match > 0.65 AND novelty > 0.5 AND !duplication_risk
         | med  if semantic_match > 0.45 AND novelty > 0.3 AND !duplication_risk
         | low  if semantic_match > 0.25 OR (novelty > 0.7 AND duplication_risk)
         | none otherwise
