+/** Filesystem-safe node id fragment (letters/digits/dash/underscore). */
+export function safeId(raw) {
+  const s = String(raw == null ? "" : raw)
+    .toLowerCase()
+    .replace(/[^a-z0-9_-]+/g, "-")
+    .replace(/-+/g, "-")
+    .replace(/^[-_]+|[-_]+$/g, "")
+    .slice(0, 80);
+  if (!s || s.includes("..")) return "x";
