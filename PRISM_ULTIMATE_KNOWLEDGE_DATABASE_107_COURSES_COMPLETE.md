## 14.1 Christofides Algorithm (TSP - MIT 18.433)
```javascript
function christofides(graph) {
    // 1. Find Minimum Spanning Tree
    const mst = primMST(graph);
    
    // 2. Find odd-degree vertices
    const oddVertices = findOddDegreeVertices(mst);
    
    // 3. Minimum weight perfect matching on odd vertices
    const matching = minimumWeightMatching(graph, oddVertices);
    
    // 4. Combine MST and matching to get Eulerian graph
    const eulerianGraph = combineGraphs(mst, matching);
    
    // 5. Find Eulerian circuit
    const circuit = hierholzer(eulerianGraph);
    
    // 6. Make Hamiltonian by shortcutting
    return makeHamiltonian(circuit);
}
// Approximation ratio: 3/2 optimal
// Time complexity: O(n³)
```

## 14.2 B+ Tree (MIT 6.830)
```javascript
class BPlusTree {
    constructor(order = 4) {
        this.root = new LeafNode();
        this.order = order;
    }
    
    search(key) {
        let node = this.root;
        while (!node.isLeaf) {
            node = node.children[this.findChildIndex(node, key)];
        }
        return node.find(key);
    }
    
    rangeSearch(low, high) {
        let node = this.findLeaf(low);
        const results = [];
        while (node && node.keys[0] <= high) {
            for (const [key, value] of node.entries()) {
                if (key >= low && key <= high) results.push(value);
            }
            node = node.next;
        }
        return results;
    }
}
```

## 14.3 Interior Point Method (MIT 6.251J)
```javascript
function interiorPointMethod(c, A, b, x0, mu = 0.1, epsilon = 1e-6) {
    let x = x0;
    let t = 1;
    
    while (c.length / t > epsilon) {
        // Newton step for barrier problem
        x = newtonStep(x, t, c, A, b);
        t = t / mu;  // Increase t (decrease barrier)
    }
    return x;
}

function barrierFunction(x) {
    return -x.reduce((sum, xi) => sum + Math.log(xi), 0);
}
```

## 14.4 Extended Kalman Filter (MIT 2.004)
```javascript
class ExtendedKalmanFilter {
    constructor(f, h, F, H, Q, R) {
        this.f = f;  // State transition (nonlinear)
        this.h = h;  // Measurement (nonlinear)
        this.F = F;  // Jacobian of f
        this.H = H;  // Jacobian of h
        this.Q = Q;  // Process noise
        this.R = R;  // Measurement noise
    }
    
    update(x, P, z) {
        // Predict
        const x_pred = this.f(x);
        const F = this.F(x);
        const P_pred = F.multiply(P).multiply(F.T()).add(this.Q);
        
        // Update
        const H = this.H(x_pred);
        const y = z.subtract(this.h(x_pred));
        const S = H.multiply(P_pred).multiply(H.T()).add(this.R);
        const K = P_pred.multiply(H.T()).multiply(S.inverse());
        
        const x_new = x_pred.add(K.multiply(y));
        const P_new = P_pred.subtract(K.multiply(H).multiply(P_pred));
        
        return [x_new, P_new];
    }
}
```

## 14.5 Voronoi Diagram - Fortune's Algorithm (MIT 18.086)
```javascript
class FortuneAlgorithm {
    compute(sites) {
        this.events = new PriorityQueue();
        this.beachline = new BeachLine();
        this.voronoi = new VoronoiDiagram();
        
        // Initialize with site events
        sites.forEach(s => this.events.push({ type: 'site', point: s }));
        
        while (!this.events.isEmpty()) {
            const event = this.events.pop();
            if (event.type === 'site') {
                this.handleSiteEvent(event.point);
            } else {
                this.handleCircleEvent(event);
            }
        }
        return this.voronoi;
    }
    
    handleSiteEvent(site) {
        const arc = this.beachline.findArcAbove(site);
        this.beachline.insertArc(site, arc);
        this.checkCircleEvent(/* neighbors */);
    }
}
// Time complexity: O(n log n)
```

## 14.6 PID Controller (MIT 2.14)
```javascript
class PIDController {
    constructor(Kp, Ki, Kd, dt = 0.001) {
        this.Kp = Kp;
        this.Ki = Ki;
        this.Kd = Kd;
        this.dt = dt;
        this.integral = 0;
        this.prevError = 0;
    }
    
    compute(setpoint, measured) {
        const error = setpoint - measured;
        
        // Proportional
        const P = this.Kp * error;
        
        // Integral with anti-windup
        this.integral += error * this.dt;
        this.integral = Math.max(-100, Math.min(100, this.integral));
        const I = this.Ki * this.integral;
        
        // Derivative with filtering
        const derivative = (error - this.prevError) / this.dt;
        const D = this.Kd * derivative;
        this.prevError = error;
        
        return P + I + D;
    }
}
```

```javascript
// Specific Cutting Force (Kienzle)

Fc = Kc × A = Kc × ap × f
```

## 16.1 Bryan's Five Principles

### Principle 1: Use Direct Measurement
- Measure at the functional point
- Avoid intermediate transformations
- Minimize measurement chain length

### Principle 2: Abbe Principle
```
Error = Offset × Angular_Error
```
- Align measurement axis with functional point
- Minimize offset distance

### Principle 3: Structural Loop
- Keep loop short and stiff
- Target stiffness: >500 N/µm for ultra-precision
- Use symmetric designs

### Principle 4: Thermal Control
```
ΔL = L₀ × α × ΔT
```
Thermal expansion coefficients:
- Steel: 11.7 µm/m/°C
- Aluminum: 23.1 µm/m/°C  
- Granite: 5-8 µm/m/°C
- Invar: 1.2 µm/m/°C
- Zerodur: 0.02 µm/m/°C

### Principle 5: Stable Reference
- Use thermally stable materials
- Granite surface plates recommended
- Temperature-controlled environments

## Denavit-Hartenberg Parameters

```javascript
class DHParameters {
    constructor(alpha, a, d, theta) {
        this.alpha = alpha;  // Link twist
        this.a = a;          // Link length
        this.d = d;          // Link offset
        this.theta = theta;  // Joint angle
    }
    
    getTransformMatrix() {
        const c_t = Math.cos(this.theta);
        const s_t = Math.sin(this.theta);
        const c_a = Math.cos(this.alpha);
        const s_a = Math.sin(this.alpha);
        
        return [
            [c_t, -s_t, 0, this.a],
            [s_t * c_a, c_t * c_a, -s_a, -s_a * this.d],
            [s_t * s_a, c_t * s_a, c_a, c_a * this.d],
            [0, 0, 0, 1]
        ];
```

### **Tool Deflection**

```
CANTILEVER BEAM (End Mill)
─────────────────────────────────────────────────
δ = (F × L³) / (3 × E × I)

Where:
  δ = deflection at tip
  F = force
  L = overhang length
  E = Young's modulus
  I = second moment of area

For circular cross-section:
I = π × d⁴ / 64

Minimize deflection:
  1. Reduce L (shorter overhang)
  2. Increase d (larger diameter)
  3. Higher E (stiffer material)
  4. Reduce F (lighter cuts)

Example:
Tool: d=10mm, L=100mm, E=600 GPa (carbide)
Force: F=500N
I = π × 10⁴ / 64 = 490.9 mm⁴
δ = (500 × 100³) / (3 × 600e6 × 490.9)
δ = 0.566 mm (unacceptable!)

Reduce to L=50mm:
δ = 0.071 mm (better)
```

```javascript
        const a0 = u0;
        const a1 = u0_dot;
        const a2 = (3*(uf - u0) - 2*u0_dot*tf - uf_dot*tf) / (tf*tf);

    // (Common in CNC - G1 moves with acceleration limits)

        const t_blend = tf / 2 - Math.sqrt((tf*tf)/4 - (uf - u0) / acc);
        
        // Three phases: acceleration, constant velocity, deceleration
        const u_blend = u0 + 0.5 * acc * t_blend * t_blend;

        return {
            evaluate: (t) => {

                    // Acceleration phase
                    return u0 + 0.5 * acc * t * t;
```

```javascript
class LinearRegression {
```

```javascript
            const h = this.hypothesis(theta, X[i]);
            sum += y[i] * Math.log(h) + (1 - y[i]) * Math.log(1 - h);
        }

    }

            const h = this.hypothesis(theta, X[i]);
            const error = h - y[i];
            
            for (let j = 0; j < n; j++) {
                grad[j] += error * X[i][j];
            }
        }
        
        return grad.map(g => g / m);
    }
    
    // Prediction
    predict(theta, x, threshold = 0.5) {
        return this.hypothesis(theta, x) >= threshold ? 1 : 0;
```

```javascript
class SVM {
    
    // Functional margin: Î³Ì‚â± = yâ±(wáµ€xâ± + b)
    functionalMargin(w, b, x, y) {
        return y * (dot(w, x) + b);
    }
    
    // Geometric margin: Î³â± = yâ±((w/||w||)áµ€xâ± + b/||w||)
    geometricMargin(w, b, x, y) {
        const w_norm = norm(w);
        return y * (dot(w, x) / w_norm + b / w_norm);
    }
    
    // Optimal margin classifier
    // max Î³ s.t. yâ±(wáµ€xâ± + b) â‰¥ Î³, ||w|| = 1
    // Equivalent to: min (1/2)||w||Â² s.t. yâ±(wáµ€xâ± + b) â‰¥ 1
    
    // Kernel functions
    kernels = {
        linear: (x1, x2) => dot(x1, x2),
        
        polynomial: (x1, x2, d = 3) => Math.pow(1 + dot(x1, x2), d),
        
        rbf: (x1, x2, sigma = 1) => {
            const diff = subtract(x1, x2);
            return Math.exp(-dot(diff, diff) / (2 * sigma * sigma));
        },
        
        sigmoid: (x1, x2, kappa = 0.01, c = 0) => 
            Math.tanh(kappa * dot(x1, x2) + c)
    };
    
    // SMO Algorithm (Sequential Minimal Optimization)
    smo(X, y, C, kernel, tol = 1e-3, maxPasses = 100) {
        const m = X.length;
        const alpha = new Array(m).fill(0);
        let b = 0;
        let passes = 0;
        
        while (passes < maxPasses) {
            let numChangedAlphas = 0;
            
            for (let i = 0; i < m; i++) {
                const Ei = this.computeError(alpha, b, X, y, i, kernel);
                
                if ((y[i] * Ei < -tol && alpha[i] < C) ||
                    (y[i] * Ei > tol && alpha[i] > 0)) {
                    
                    // Select j â‰  i randomly
                    let j = Math.floor(Math.random() * m);
                    while (j === i) j = Math.floor(Math.random() * m);
                    
                    const Ej = this.computeError(alpha, b, X, y, j, kernel);
                    
                    // Save old alphas
                    const alpha_i_old = alpha[i];
                    const alpha_j_old = alpha[j];
                    
                    // Compute bounds
                    let L, H;
                    if (y[i] !== y[j]) {
                        L = Math.max(0, alpha[j] - alpha[i]);
                        H = Math.min(C, C + alpha[j] - alpha[i]);
                    } else {
                        L = Math.max(0, alpha[i] + alpha[j] - C);
                        H = Math.min(C, alpha[i] + alpha[j]);
                    }
                    
                    if (L === H) continue;
                    
                    // Compute eta
                    const eta = 2 * kernel(X[i], X[j]) - 
                                kernel(X[i], X[i]) - 
                                kernel(X[j], X[j]);
                    
                    if (eta >= 0) continue;
                    
                    // Update alpha_j
                    alpha[j] = alpha[j] - y[j] * (Ei - Ej) / eta;
                    alpha[j] = Math.max(L, Math.min(H, alpha[j]));
                    
                    if (Math.abs(alpha[j] - alpha_j_old) < 1e-5) continue;
                    
                    // Update alpha_i
                    alpha[i] = alpha[i] + y[i] * y[j] * (alpha_j_old - alpha[j]);
                    
                    // Update b
                    const b1 = b - Ei - y[i] * (alpha[i] - alpha_i_old) * kernel(X[i], X[i])
                                      - y[j] * (alpha[j] - alpha_j_old) * kernel(X[i], X[j]);
                    const b2 = b - Ej - y[i] * (alpha[i] - alpha_i_old) * kernel(X[i], X[j])
                                      - y[j] * (alpha[j] - alpha_j_old) * kernel(X[j], X[j]);
                    
                    if (0 < alpha[i] && alpha[i] < C) b = b1;
                    else if (0 < alpha[j] && alpha[j] < C) b = b2;
                    else b = (b1 + b2) / 2;
                    
                    numChangedAlphas++;
                }
            }
            
            if (numChangedAlphas === 0) passes++;
            else passes = 0;
        }
        
        return { alpha, b };
    }
}
```

```javascript
// From CS 229 Notes 7a
class KMeans {
    
    // Initialize centroids randomly
    initCentroids(X, k) {
        const indices = [];
        while (indices.length < k) {
            const idx = Math.floor(Math.random() * X.length);
            if (!indices.includes(idx)) indices.push(idx);
        }
        return indices.map(i => [...X[i]]);
    }
    
    // Assign each point to closest centroid
    assignClusters(X, centroids) {
        return X.map(x => {
            let minDist = Infinity;
            let cluster = 0;
            
            for (let j = 0; j < centroids.length; j++) {
                const dist = this.distance(x, centroids[j]);
                if (dist < minDist) {
                    minDist = dist;
                    cluster = j;
                }
            }
            
            return cluster;
        });
    }
    
    // Update centroids as mean of assigned points
    updateCentroids(X, assignments, k) {
        const centroids = [];
        
        for (let j = 0; j < k; j++) {
            const points = X.filter((_, i) => assignments[i] === j);
            if (points.length > 0) {
                centroids.push(this.mean(points));
            } else {
                // Random reinitialization for empty cluster
                centroids.push([...X[Math.floor(Math.random() * X.length)]]);
            }
        }
        
        return centroids;
    }
    
    // Main algorithm
    fit(X, k, maxIter = 100) {
        let centroids = this.initCentroids(X, k);
        let assignments = [];
        
        for (let iter = 0; iter < maxIter; iter++) {
            const newAssignments = this.assignClusters(X, centroids);
            centroids = this.updateCentroids(X, newAssignments, k);
            
            // Check convergence
            if (JSON.stringify(assignments) === JSON.stringify(newAssignments)) {
                break;
            }
            assignments = newAssignments;
        }
        
        return { centroids, assignments };
    }
    
    // Distortion: J = Î£ ||xâ½â±â¾ - Î¼_c(i)||Â²
    distortion(X, centroids, assignments) {
        return X.reduce((sum, x, i) => {
            return sum + this.distance(x, centroids[assignments[i]]) ** 2;
        }, 0);
    }
    
    distance(a, b) {
        return Math.sqrt(a.reduce((sum, val, i) => sum + (val - b[i]) ** 2, 0));
    }
    
    mean(points) {
        const n = points[0].length;
        const sum = new Array(n).fill(0);
        
        for (const p of points) {
            for (let i = 0; i < n; i++) {
                sum[i] += p[i];
            }
        }
        
        return sum.map(s => s / points.length);
    }
}
```

```javascript
// From CS 229 Notes 8
class GaussianMixture {
    
    // E-step: Compute responsibilities
    estep(X, phi, mu, sigma) {
        const k = phi.length;
        const n = X.length;
        const w = new Array(n).fill(null).map(() => new Array(k).fill(0));
        
        for (let i = 0; i < n; i++) {
            let sum = 0;
            
            for (let j = 0; j < k; j++) {
                w[i][j] = phi[j] * this.gaussianPDF(X[i], mu[j], sigma[j]);
                sum += w[i][j];
            }
            
            // Normalize
            for (let j = 0; j < k; j++) {
                w[i][j] /= sum;
            }
        }
        
        return w;
    }
    
    // M-step: Update parameters
    mstep(X, w) {
        const k = w[0].length;
        const n = X.length;
        const d = X[0].length;
        
        const phi = new Array(k).fill(0);
        const mu = new Array(k).fill(null).map(() => new Array(d).fill(0));
        const sigma = new Array(k).fill(null).map(() => 
            new Array(d).fill(null).map(() => new Array(d).fill(0))
        );
        
        for (let j = 0; j < k; j++) {
            let w_sum = 0;
            
            for (let i = 0; i < n; i++) {
                w_sum += w[i][j];
                
                for (let l = 0; l < d; l++) {
                    mu[j][l] += w[i][j] * X[i][l];
                }
            }
            
            phi[j] = w_sum / n;
            
            for (let l = 0; l < d; l++) {
                mu[j][l] /= w_sum;
            }
            
            // Update covariance
            for (let i = 0; i < n; i++) {
                const diff = subtract(X[i], mu[j]);
                const outer = outerProduct(diff, diff);
                
                for (let l1 = 0; l1 < d; l1++) {
                    for (let l2 = 0; l2 < d; l2++) {
                        sigma[j][l1][l2] += w[i][j] * outer[l1][l2];
                    }
                }
            }
            
            for (let l1 = 0; l1 < d; l1++) {
                for (let l2 = 0; l2 < d; l2++) {
                    sigma[j][l1][l2] /= w_sum;
                }
            }
        }
        
        return { phi, mu, sigma };
    }
    
    // Multivariate Gaussian PDF
    gaussianPDF(x, mu, sigma) {
        const d = x.length;
        const diff = subtract(x, mu);
        const sigma_inv = inverse(sigma);
        const det = determinant(sigma);
        
        const exponent = -0.5 * dot(diff, multiply(sigma_inv, diff));
        const coef = 1 / Math.sqrt(Math.pow(2 * Math.PI, d) * det);
        
        return coef * Math.exp(exponent);
    }
    
    // Full EM algorithm
    fit(X, k, maxIter = 100, tol = 1e-6) {
        // Initialize
        let phi = new Array(k).fill(1 / k);
        let mu = this.initMeans(X, k);
        let sigma = this.initCovariances(X, k);
        
        let prevLogLikelihood = -Infinity;
        
        for (let iter = 0; iter < maxIter; iter++) {
            // E-step
            const w = this.estep(X, phi, mu, sigma);
            
            // M-step
            ({ phi, mu, sigma } = this.mstep(X, w));
            
            // Check convergence
            const logLikelihood = this.logLikelihood(X, phi, mu, sigma);
            if (Math.abs(logLikelihood - prevLogLikelihood) < tol) {
                break;
            }
            prevLogLikelihood = logLikelihood;
        }
        
        return { phi, mu, sigma };
    }
}
```

```javascript
// From CS 229 Notes 12
class ReinforcementLearning {
    
    // MDP: (S, A, P_sa, Î³, R)
    constructor(states, actions, transitions, gamma, rewards) {
        this.S = states;
        this.A = actions;
        this.P = transitions;  // P[s][a][s'] = probability
        this.gamma = gamma;
        this.R = rewards;      // R[s] or R[s][a]
    }
    
    // Value Iteration
    valueIteration(epsilon = 1e-6, maxIter = 1000) {
        const V = {};
        for (const s of this.S) V[s] = 0;
        
        for (let iter = 0; iter < maxIter; iter++) {
            let delta = 0;
            
            for (const s of this.S) {
                const v = V[s];
                
                // V(s) = max_a [R(s,a) + Î³ Î£ P(s'|s,a)V(s')]
                let maxValue = -Infinity;
                
                for (const a of this.A) {
                    let value = this.R[s];
                    
                    for (const s_prime of this.S) {
                        value += this.gamma * this.P[s][a][s_prime] * V[s_prime];
                    }
                    
                    maxValue = Math.max(maxValue, value);
                }
                
                V[s] = maxValue;
                delta = Math.max(delta, Math.abs(v - V[s]));
            }
            
            if (delta < epsilon) break;
        }
        
        return V;
    }
    
    // Extract optimal policy from value function
    extractPolicy(V) {
        const policy = {};
        
        for (const s of this.S) {
            let bestAction = null;
            let bestValue = -Infinity;
            
            for (const a of this.A) {
                let value = this.R[s];
                
                for (const s_prime of this.S) {
                    value += this.gamma * this.P[s][a][s_prime] * V[s_prime];
                }
                
                if (value > bestValue) {
                    bestValue = value;
                    bestAction = a;
                }
            }
            
            policy[s] = bestAction;
        }
        
        return policy;
    }
    
    // Q-Learning (model-free)
    qLearning(episodes, alpha = 0.1, epsilon = 0.1) {
        const Q = {};
        for (const s of this.S) {
            Q[s] = {};
            for (const a of this.A) Q[s][a] = 0;
        }
        
        for (let ep = 0; ep < episodes; ep++) {
            let s = this.randomState();
            
            while (!this.isTerminal(s)) {
                // Îµ-greedy action selection
                let a;
                if (Math.random() < epsilon) {
                    a = this.randomAction();
                } else {
                    a = this.argmax(Q[s]);
                }
                
                // Take action, observe reward and next state
                const { s_prime, reward } = this.step(s, a);
                
                // Q-learning update
                const maxQ = Math.max(...Object.values(Q[s_prime]));
                Q[s][a] = Q[s][a] + alpha * (reward + this.gamma * maxQ - Q[s][a]);
                
                s = s_prime;
            }
        }
        
        return Q;
    }
}
```

```javascript
// From CS 229 Notes 10
class PCA {
    
    // Preprocessing: normalize mean and variance
    preprocess(X) {
        const n = X.length;
        const d = X[0].length;
        
        // Compute mean
        const mu = new Array(d).fill(0);
        for (const x of X) {
            for (let j = 0; j < d; j++) {
                mu[j] += x[j];
            }
        }
        for (let j = 0; j < d; j++) mu[j] /= n;
        
        // Center data
        const X_centered = X.map(x => x.map((val, j) => val - mu[j]));
        
        // Compute variance
        const sigma = new Array(d).fill(0);
        for (const x of X_centered) {
            for (let j = 0; j < d; j++) {
                sigma[j] += x[j] * x[j];
            }
        }
        for (let j = 0; j < d; j++) sigma[j] = Math.sqrt(sigma[j] / n);
        
        // Normalize
        const X_normalized = X_centered.map(x => 
            x.map((val, j) => sigma[j] > 0 ? val / sigma[j] : val)
        );
        
        return { X: X_normalized, mu, sigma };
    }
    
    // Compute principal components
    fit(X, k) {
        const { X: X_norm, mu, sigma } = this.preprocess(X);
        
        // Compute covariance matrix
        const n = X_norm.length;
        const d = X_norm[0].length;
        const Sigma = new Array(d).fill(null).map(() => new Array(d).fill(0));
        
        for (const x of X_norm) {
            for (let i = 0; i < d; i++) {
                for (let j = 0; j < d; j++) {
                    Sigma[i][j] += x[i] * x[j];
                }
            }
        }
        
        for (let i = 0; i < d; i++) {
            for (let j = 0; j < d; j++) {
                Sigma[i][j] /= n;
            }
        }
        
        // Eigendecomposition
        const { eigenvalues, eigenvectors } = this.eigenDecomposition(Sigma);
        
        // Sort by eigenvalue (descending)
        const indices = eigenvalues.map((_, i) => i)
            .sort((a, b) => eigenvalues[b] - eigenvalues[a]);
        
        // Select top k components
        const components = indices.slice(0, k).map(i => eigenvectors[i]);
        const explainedVariance = indices.slice(0, k).map(i => eigenvalues[i]);
        const totalVariance = eigenvalues.reduce((a, b) => a + b, 0);
        
        return {
            components,
            explainedVariance,
            explainedVarianceRatio: explainedVariance.map(v => v / totalVariance),
            mu,
            sigma
        };
    }
    
    // Project data onto principal components
    transform(X, components, mu, sigma) {
        return X.map(x => {
            // Normalize
            const x_norm = x.map((val, j) => (val - mu[j]) / sigma[j]);
            
            // Project
            return components.map(pc => dot(x_norm, pc));
        });
    }
    
    // Reconstruct from reduced representation
    inverse_transform(Z, components, mu, sigma) {
        return Z.map(z => {
            // Combine principal components
            const x_norm = new Array(components[0].length).fill(0);
            for (let i = 0; i < z.length; i++) {
                for (let j = 0; j < x_norm.length; j++) {
                    x_norm[j] += z[i] * components[i][j];
                }
            }
            
            // Denormalize
            return x_norm.map((val, j) => val * sigma[j] + mu[j]);
        });
    }
}
```

#### **PID Controller Implementation:**
```python
class PIDController:
    def __init__(self, Kp, Ki, Kd, dt=0.001):
        self.Kp = Kp
        self.Ki = Ki
        self.Kd = Kd
        self.dt = dt
        self.integral = 0
        self.prev_error = 0
    
    def update(self, setpoint, measured):
        error = setpoint - measured
        
        # Proportional term
        P = self.Kp * error
        
        # Integral term with anti-windup
        self.integral = np.clip(
            self.integral + error * self.dt,
            -100, 100  # Anti-windup limits
        )
        I = self.Ki * self.integral
        
        # Derivative term with filtering
        derivative = (error - self.prev_error) / self.dt
        D = self.Kd * derivative
        
        self.prev_error = error
        
        # Control output
        return P + I + D
```

#### **Kalman Filter for State Estimation:**
```python
class KalmanFilter:
    def __init__(self, F, H, Q, R, P0, x0):
        self.F = F  # State transition
        self.H = H  # Measurement matrix
        self.Q = Q  # Process noise
        self.R = R  # Measurement noise
        self.P = P0 # Error covariance
        self.x = x0 # Initial state
    
    def predict(self, u=None):
        # State prediction
        self.x = self.F @ self.x
        if u is not None:
            self.x += self.B @ u
        
        # Covariance prediction
        self.P = self.F @ self.P @ self.F.T + self.Q
    
    def update(self, z):
        # Kalman gain
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        
        # State update
        y = z - self.H @ self.x  # Residual
        self.x = self.x + K @ y
        
        # Covariance update
        I = np.eye(len(self.x))
        self.P = (I - K @ self.H) @ self.P
        
        return self.x
```

### **2.5 OPTIMIZATION ALGORITHMS (MIT 18.433, 6.251j)**

#### **Simplex Algorithm:**
```python
def simplex(c, A, b):
    """
    Solve: minimize c^T x
    Subject to: Ax <= b, x >= 0
    """
    m, n = A.shape
    
    # Add slack variables
    tableau = np.zeros((m+1, n+m+1))
    tableau[:-1, :n] = A
    tableau[:-1, n:n+m] = np.eye(m)
    tableau[:-1, -1] = b
    tableau[-1, :n] = -c
    
    # Basic variables
    basis = list(range(n, n+m))
    
    while np.any(tableau[-1, :-1] < 0):
        # Entering variable (most negative)
        col = np.argmin(tableau[-1, :-1])
        
        # Leaving variable (minimum ratio test)
        ratios = tableau[:-1, -1] / tableau[:-1, col]
        ratios[ratios <= 0] = np.inf
        row = np.argmin(ratios)
        
        # Pivot
        tableau[row] /= tableau[row, col]
        for i in range(m+1):
            if i != row:
                tableau[i] -= tableau[i, col] * tableau[row]
        
        basis[row] = col
    
    # Extract solution
    x = np.zeros(n)
    for i, var in enumerate(basis):
        if var < n:
            x[var] = tableau[i, -1]
    
    return x, -tableau[-1, -1]
```

#### **Network Flow (Ford-Fulkerson):**
```python
def max_flow(graph, source, sink):
    """
    Find maximum flow using Ford-Fulkerson
    """
    def bfs(graph, source, sink, parent):
        visited = set([source])
        queue = [source]
        
        while queue:
            u = queue.pop(0)
            for v in graph[u]:
                if v not in visited and graph[u][v] > 0:
                    visited.add(v)
                    queue.append(v)
                    parent[v] = u
                    if v == sink:
                        return True
        return False
    
    parent = {}
    max_flow_value = 0
    
    while bfs(graph, source, sink, parent):
        # Find minimum residual capacity
        path_flow = float('inf')
        s = sink
        while s != source:
            path_flow = min(path_flow, graph[parent[s]][s])
            s = parent[s]
        
        # Update residual capacities
        v = sink
        while v != source:
            u = parent[v]
            graph[u][v] -= path_flow
            graph[v][u] += path_flow
            v = parent[v]
        
        max_flow_value += path_flow
        parent = {}
    
    return max_flow_value
```
