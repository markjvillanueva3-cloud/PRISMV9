```powershell
# CPU load (average over 5 s)
$cpu = Get-Counter '\Processor(_Total)\% Processor Time' | Select-Object -ExpandProperty CounterSamples | Measure-Object -Average | Select-Object -ExpandProperty Average

# RAM usage
$mem = Get-WmiObject Win32_OperatingSystem
$ramUsedPct = ($mem.TotalVisibleMemorySize - $mem.FreePhysicalMemory) / $mem.TotalVisibleMemorySize * 100

# GPU VRAM (NVIDIA)
$gpuVram = nvidia-smi --query-gpu=memory.used,memory.total --format=csv,noheader,nounits | ForEach-Object {
    $parts = $_ -split ','
    [int]$used = $parts[0].Trim()
    [int]$total = $parts[1].Trim()
    ($used / $total) * 100
} | Measure-Object -Average | Select-Object -ExpandProperty Average

# Process counts
$nodeCnt = (Get-Process node).Count
$bashCnt = (Get-Process bash).Count
```
